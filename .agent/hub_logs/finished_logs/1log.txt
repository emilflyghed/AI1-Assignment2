```python
import random

class HangedManGame:
    def __init__(self, words):
        self.words = words
        self.secret_word = self._choose_word()
        self.guessed_letters = set()
        self.attempts_left = 6  # Standard for Hanged Man
        self.display_word = ["_" for _ in self.secret_word]

    def _choose_word(self):
        return random.choice(self.words).upper()

    def guess(self, letter):
        letter = letter.upper()
        if not letter.isalpha() or len(letter) != 1:
            return "Invalid input. Please guess a single letter."

        if letter in self.guessed_letters:
            return f"You already guessed '{letter}'. Try again."

        self.guessed_letters.add(letter)

        if letter in self.secret_word:
            for i, char in enumerate(self.secret_word):
                if char == letter:
                    self.display_word[i] = letter
            if "_" not in self.display_word:
                return "win"
            return f"Good guess! {' '.join(self.display_word)}"
        else:
            self.attempts_left -= 1
            if self.attempts_left == 0:
                return "lose"
            return f"Wrong guess. Attempts left: {self.attempts_left}. {' '.join(self.display_word)}"

    def get_display_word(self):
        return " ".join(self.display_word)

    def get_attempts_left(self):
        return self.attempts_left

    def get_guessed_letters(self):
        return sorted(list(self.guessed_letters))

    def is_game_over(self):
        return self.attempts_left == 0 or "_" not in self.display_word
```

```python
import random

class HangedMan:
    def __init__(self):
        self.words = ['python', 'hangman', 'challenge', 'programming']
        self.word = random.choice(self.words)
        self.guesses = []
        self.max_attempts = 6

    def display_word(self):
        display = ''.join([letter if letter in self.guesses else '_' for letter in self.word])
        return display

    def make_guess(self, letter):
        if letter in self.guesses:
            return 'Already guessed!'
        self.guesses.append(letter)
        if letter not in self.word:
            self.max_attempts -= 1
            return 'Wrong guess!'
        return 'Good guess!'

    def is_game_over(self):
        if self.max_attempts <= 0:
            return True
        if all(letter in self.guesses for letter in self.word):
            return True
        return False

    def play(self):
        print('Welcome to Hanged Man!')
        while not self.is_game_over():
            print('\nCurrent word: ' + self.display_word())
            print('Attempts remaining: ' + str(self.max_attempts))
            guess = input('Guess a letter: ').lower()
            result = self.make_guess(guess)
            print(result)
        if self.max_attempts > 0:
            print(f'Congratulations! You guessed the word: {self.word}')
        else:
            print(f'Game over! The word was: {self.word}')

if __name__ == '__main__':
    game = HangedMan()
    game.play()
```

```python
MAX_WRONG = 6

class HangmanGame:
    def __init__(self, word: str):
        self.word = word.upper()
        self.guessed_letters = set()
        self.wrong_guesses = 0

    def guess(self, letter: str):
        letter = letter.upper()
        if letter in self.guessed_letters:
            return None  # Already guessed
        self.guessed_letters.add(letter)
        if letter in self.word:
            return True
        else:
            self.wrong_guesses += 1
            return False

    def get_display_word(self) -> str:
        return " ".join(
            letter if letter in self.guessed_letters else "_"
            for letter in self.word
        )

    def is_won(self) -> bool:
        return all(letter in self.guessed_letters for letter in self.word)

    def is_lost(self) -> bool:
        return self.wrong_guesses >= MAX_WRONG

    def get_wrong_letters(self) -> list:
        return sorted([l for l in self.guessed_letters if l not in self.word])

    def get_hangman_art(self) -> str:
        return HANGMAN_STAGES[self.wrong_guesses]

    def get_status(self) -> dict:
        return {
            "display_word": self.get_display_word(),
            "wrong_guesses": self.wrong_guesses,
            "max_wrong": MAX_WRONG,
            "wrong_letters": self.get_wrong_letters(),
            "guessed_letters": sorted(self.guessed_letters),
            "is_won": self.is_won(),
            "is_lost": self.is_lost(),
            "hangman_art": self.get_hangman_art(),
        }


def play_game(word: str):
    game = HangmanGame(word)
    print("\n🎮 Welcome to Hangman!\n")
    while not game.is_won() and not game.is_lost():
        status = game.get_status()
        print(status["hangman_art"])
        print(f"Word: {status['display_word']}")
        print(f"Wrong guesses ({status['wrong_guesses']}/{status['max_wrong']}): {', '.join(status['wrong_letters']) or 'none'}")
        print()
        guess = input("Guess a letter: ").strip()
        if len(guess) != 1 or not guess.isalpha():
            print("⚠️  Please enter a single letter.\n")
            continue
        result = game.guess(guess)
        if result is None:
            print(f"⚠️  You already guessed '{guess.upper()}'!\n")
        elif result:
            print(f"✅ '{guess.upper()}' is in the word!\n")
        else:
            print(f"❌ '{guess.upper()}' is NOT in the word.\n")
    status = game.get_status()
    print(status["hangman_art"])
    if game.is_won():
        print(f"🎉 You won! The word was: {game.word}")
    else:
        print(f"💀 Game over! The word was: {game.word}")
```
