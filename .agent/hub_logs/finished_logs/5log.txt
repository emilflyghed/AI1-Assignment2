# Basic Calculator - supports +, -, *, /, **, % and parentheses, sqrt, pi

import math, re

def calculate(expression: str) -> str:
    """Evaluate a math expression string safely and return the result."""
    expression = expression.strip()
    if not expression:
        return "Error: Empty input"
    if not re.match(r'^[\d\s\+\-\*\/\.\%\^\(\)a-zA-Z_]+$', expression):
        return "Error: Invalid characters in expression"
    expression = expression.replace("^", "**")
    safe_globals = {"__builtins__": {}}
    safe_locals = {"sqrt": math.sqrt, "pi": math.pi, "abs": abs}
    try:
        result = eval(expression, safe_globals, safe_locals)
        if result in (float('inf'), float('-inf')):
            return "Error: Division by zero"
        return f"= {result}"
    except ZeroDivisionError:
        return "Error: Division by zero"
    except SyntaxError:
        return "Error: Invalid expression syntax"
    except Exception as e:
        return f"Error: {e}"

def main():
    print("Basic Calculator (type 'q' to quit)")
    print("Supports: +  -  *  /  %  ^ (power)  sqrt(x)  abs(x)  pi  parentheses")
    while True:
        user_input = input("\n> ").strip()
        if user_input.lower() == 'q':
            print("Goodbye!")
            break
        print(calculate(user_input))

if __name__ == "__main__":
    main()
